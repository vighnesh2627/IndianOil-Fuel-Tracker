import tkinter as tk
from tkinter import ttk, messagebox
import sqlite3
from datetime import date

# Database
db = sqlite3.connect("fuel.db")
cur = db.cursor()

cur.execute("""
CREATE TABLE IF NOT EXISTS fuel(
    id INTEGER PRIMARY KEY,
    date TEXT,
    vehicle TEXT,
    km REAL,
    litres REAL,
    price REAL,
    cost REAL
)
""")
db.commit()


# Functions
def calculate():
    try:
        cost = float(litres.get()) * float(price.get())
        cost_var.set("₹" + str(round(cost, 2)))
    except:
        messagebox.showerror("Error", "Enter valid numbers")


def save():
    try:
        d = date_var.get()
        v = vehicle.get()
        k = float(km.get())
        l = float(litres.get())
        p = float(price.get())

        if d == "" or v == "" or k < 0 or l <= 0 or p <= 0:
            raise ValueError

        cur.execute(
            "INSERT INTO fuel(date,vehicle,km,litres,price,cost) VALUES(?,?,?,?,?,?)",
            (d, v, k, l, p, l * p)
        )
        db.commit()

        messagebox.showinfo("Saved", "Fuel entry saved")
        clear()
        show_records()
        summary()

    except:
        messagebox.showerror("Error", "Enter valid details")


def clear():
    date_var.set(str(date.today()))
    vehicle.set("")
    km.set("")
    litres.set("")
    price.set("")
    cost_var.set("₹0.00")


def show_records():
    for x in table.get_children():
        table.delete(x)

    cur.execute("SELECT * FROM fuel ORDER BY id DESC")

    for r in cur.fetchall():
        table.insert("", "end", values=r)


def delete():
    selected = table.selection()

    if not selected:
        messagebox.showwarning("Delete", "Select a record")
        return

    for x in selected:
        rid = table.item(x)["values"][0]
        cur.execute("DELETE FROM fuel WHERE id=?", (rid,))

    db.commit()
    show_records()
    summary()


def summary():
    # Total expense and fuel
    cur.execute("SELECT SUM(cost), SUM(litres) FROM fuel")
    total, fuel = cur.fetchone()

    total = total or 0
    fuel = fuel or 0

    expense.set("₹" + format(total, ".2f"))
    total_fuel.set(format(fuel, ".2f") + " L")

    # Mileage calculation
    cur.execute(
        "SELECT vehicle, km, litres FROM fuel ORDER BY vehicle, id"
    )

    rows = cur.fetchall()

    mileages = []
    total_distance = 0
    previous = {}

    for v, k, l in rows:
        if v in previous:
            distance = k - previous[v]

            if distance > 0:
                mileages.append(distance / l)
                total_distance += distance

        previous[v] = k

    # Average Mileage
    if mileages:
        avg = sum(mileages) / len(mileages)
        mileage.set(format(avg, ".2f") + " km/L")
    else:
        mileage.set("0.00 km/L")

    # Average Cost per KM
    if total_distance > 0:
        average_cost_km = total / total_distance
        costkm.set("₹" + format(average_cost_km, ".2f") + "/km")
    else:
        costkm.set("₹0.00/km")
